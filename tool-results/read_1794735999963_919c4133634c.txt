     1→'use client'
     2→
     3→import { useCallback, useEffect, useState } from 'react'
     4→import {
     5→  Cloud,
     6→  DatabaseBackup,
     7→  Download,
     8→  Github,
     9→  HardDrive,
    10→  Loader2,
    11→  PackageOpen,
    12→  RotateCcw,
    13→  ShieldCheck,
    14→  Trash2,
    15→  TriangleAlert,
    16→  UploadCloud,
    17→} from 'lucide-react'
    18→import { Card } from '@/components/ui/card'
    19→import { Button } from '@/components/ui/button'
    20→import { Badge } from '@/components/ui/badge'
    21→import { Input } from '@/components/ui/input'
    22→import { Skeleton } from '@/components/ui/skeleton'
    23→import {
    24→  Table,
    25→  TableBody,
    26→  TableCell,
    27→  TableHead,
    28→  TableHeader,
    29→  TableRow,
    30→} from '@/components/ui/table'
    31→import {
    32→  AlertDialog,
    33→  AlertDialogAction,
    34→  AlertDialogCancel,
    35→  AlertDialogContent,
    36→  AlertDialogDescription,
    37→  AlertDialogFooter,
    38→  AlertDialogHeader,
    39→  AlertDialogTitle,
    40→  AlertDialogTrigger,
    41→} from '@/components/ui/alert-dialog'
    42→import { toast } from 'sonner'
    43→import type { BackupBucketItem, BackupListResponse, BackupLocal } from '@/lib/types'
    44→import { formatBytes, formatTime } from '@/lib/format'
    45→
    46→type ListData = BackupListResponse & { totalLocalBytes?: number }
    47→
    48→export function BackupPanel() {
    49→  const [data, setData] = useState<ListData | null>(null)
    50→  const [loading, setLoading] = useState(true)
    51→  const [name, setName] = useState('')
    52→  const [creating, setCreating] = useState(false)
    53→  const [busyLocalId, setBusyLocalId] = useState<number | null>(null)
    54→  const [busyBucketKey, setBusyBucketKey] = useState<string | null>(null)
    55→
    56→  const local = data?.local ?? []
    57→  const bucket = data?.bucket ?? []
    58→  const totalLocalBytes = data?.totalLocalBytes ?? local.reduce((s, b) => s + b.sizeBytes, 0)
    59→  const totalBucketBytes = bucket.reduce((s, b) => s + b.sizeBytes, 0)
    60→
    61→  const load = useCallback(async () => {
    62→    try {
    63→      const r = await fetch('/api/backups', { cache: 'no-store' })
    64→      const json = (await r.json()) as ListData
    65→      setData(json)
    66→    } catch {
    67→      toast.error('Không tải được danh sách bản sao lưu')
    68→    } finally {
    69→      setLoading(false)
    70→    }
    71→  }, [])
    72→
    73→  useEffect(() => {
    74→    load()
    75→  }, [load])
    76→
    77→  const create = async () => {
    78→    setCreating(true)
    79→    try {
    80→      const r = await fetch('/api/backups', {
    81→        method: 'POST',
    82→        headers: { 'Content-Type': 'application/json' },
    83→        body: JSON.stringify({ name: name.trim() || 'backup' }),
    84→      })
    85→      const json = (await r.json()) as { backup?: BackupLocal; error?: string }
    86→      if (!r.ok || json.error) throw new Error(json.error || 'Tạo backup thất bại')
    87→      toast.success(
    88→        `Đã tạo backup ${json.backup?.name ?? ''} (${formatBytes(json.backup?.sizeBytes ?? 0)})`,
    89→      )
    90→      setName('')
    91→      await load()
    92→    } catch (e) {
    93→      toast.error((e as Error).message || 'Tạo backup thất bại')
    94→    } finally {
    95→      setCreating(false)
    96→    }
    97→  }
    98→
    99→  const uploadOne = async (b: BackupLocal) => {
   100→    setBusyLocalId(b.id)
   101→    toast.info(`Đang tải ${b.name} lên bucket...`)
   102→    try {
   103→      const r = await fetch('/api/backups', {
   104→        method: 'PUT',
   105→        headers: { 'Content-Type': 'application/json' },
   106→        body: JSON.stringify({ action: 'upload', id: b.id }),
   107→      })
   108→      const json = (await r.json()) as { ok?: boolean; error?: string; key?: string }
   109→      if (!r.ok || !json.ok) throw new Error(json.error || 'Tải lên bucket thất bại')
   110→      toast.success(`Đã tải lên bucket: ${json.key ?? b.name}`)
   111→      await load()
   112→    } catch (e) {
   113→      toast.error((e as Error).message || 'Tải lên bucket thất bại')
   114→    } finally {
   115→      setBusyLocalId(null)
   116→    }
   117→  }
   118→
   119→  const downloadLocal = async (b: BackupLocal) => {
   120→    setBusyLocalId(b.id)
   121→    try {
   122→      const r = await fetch('/api/backups', {
   123→        method: 'PUT',
   124→        headers: { 'Content-Type': 'application/json' },
   125→        body: JSON.stringify({ action: 'presign', id: b.id }),
   126→      })
   127→      const json = (await r.json()) as { url?: string; error?: string }
   128→      if (!r.ok || !json.url) throw new Error(json.error || 'Không tạo được link tải về')
   129→      window.open(json.url, '_blank', 'noopener')
   130→      toast.success('Đã mở link tải về (hết hạn sau 1 giờ)')
   131→    } catch (e) {
   132→      toast.error((e as Error).message || 'Không tạo được link tải về')
   133→    } finally {
   134→      setBusyLocalId(null)
   135→    }
   136→  }
   137→
   138→  const downloadBucket = async (key: string) => {
   139→    setBusyBucketKey(key)
   140→    try {
   141→      const r = await fetch('/api/backups', {
   142→        method: 'PUT',
   143→        headers: { 'Content-Type': 'application/json' },
   144→        body: JSON.stringify({ action: 'presign', key }),
   145→      })
   146→      const json = (await r.json()) as { url?: string; error?: string }
   147→      if (!r.ok || !json.url) throw new Error(json.error || 'Không tạo được link tải về')
   148→      window.open(json.url, '_blank', 'noopener')
   149→      toast.success('Đã mở link tải về (hết hạn sau 1 giờ)')
   150→    } catch (e) {
   151→      toast.error((e as Error).message || 'Không tạo được link tải về')
   152→    } finally {
   153→      setBusyBucketKey(null)
   154→    }
   155→  }
   156→
   157→  const restoreBucket = async (key: string) => {
   158→    setBusyBucketKey(key)
   159→    toast.info('Đang khôi phục backup về máy chủ...')
   160→    try {
   161→      const r = await fetch('/api/backups', {
   162→        method: 'PUT',
   163→        headers: { 'Content-Type': 'application/json' },
   164→        body: JSON.stringify({ action: 'restore', key }),
   165→      })
   166→      const json = (await r.json()) as { ok?: boolean; message?: string; error?: string }
   167→      if (!r.ok || !json.ok) throw new Error(json.error || 'Khôi phục thất bại')
   168→      toast.success(json.message || 'Đã khôi phục backup về máy chủ')
   169→    } catch (e) {
   170→      toast.error((e as Error).message || 'Khôi phục thất bại')
   171→    } finally {
   172→      setBusyBucketKey(null)
   173→    }
   174→  }
   175→
   176→  const deleteLocal = async (id: number) => {
   177→    setBusyLocalId(id)
   178→    try {
   179→      const r = await fetch(`/api/backups?type=local&id=${id}`, { method: 'DELETE' })
   180→      const json = (await r.json()) as { success?: boolean; error?: string }
   181→      if (!r.ok || !json.success) throw new Error(json.error || 'Không xóa được bản backup')
   182→      toast.success('Đã xóa bản backup trên máy chủ')
   183→      await load()
   184→    } catch (e) {
   185→      toast.error((e as Error).message || 'Không xóa được bản backup')
   186→    } finally {
   187→      setBusyLocalId(null)
   188→    }
   189→  }
   190→
   191→  const deleteBucket = async (key: string) => {
   192→    setBusyBucketKey(key)
   193→    try {
   194→      const r = await fetch(`/api/backups?type=bucket&key=${encodeURIComponent(key)}`, {
   195→        method: 'DELETE',
   196→      })
   197→      const json = (await r.json()) as { success?: boolean; error?: string }
   198→      if (!r.ok || !json.success) throw new Error(json.error || 'Không xóa được object trên bucket')
   199→      toast.success('Đã xóa object trên bucket')
   200→      await load()
   201→    } catch (e) {
   202→      toast.error((e as Error).message || 'Không xóa được object trên bucket')
   203→    } finally {
   204→      setBusyBucketKey(null)
   205→    }
   206→  }
   207→
   208→  return (
   209→    <div className="space-y-4">
   210→      {/* Hero: tạo backup + trạng thái */}
   211→      <Card className="glass rounded-2xl p-6">
   212→        <div className="flex flex-col gap-5">
   213→          <div className="flex flex-wrap items-center justify-between gap-4">
   214→            <div className="flex items-center gap-3">
   215→              <div
   216→                className="flex h-11 w-11 items-center justify-center rounded-xl border border-white/10"
   217→                style={{ background: 'color-mix(in srgb, var(--accent-1) 14%, transparent)' }}
   218→              >
   219→                <DatabaseBackup className="h-5 w-5" style={{ color: 'var(--accent-1)' }} />
   220→              </div>
   221→              <div>
   222→                <h3 className="text-base font-semibold text-zinc-100">Tạo bản sao lưu</h3>
   223→                <p className="text-xs text-zinc-500">
   224→                  Nén toàn bộ mã nguồn + cấu hình thành file .tar.gz (loại trừ node_modules,
   225→                  .next, upload, .env)
   226→                </p>
   227→              </div>
   228→            </div>
   229→            <div className="flex flex-wrap items-center gap-2">
   230→              {data ? (
   231→                data.bucketOk ? (
   232→                  <Badge className="rounded-full border-emerald-500/40 bg-emerald-500/10 text-emerald-400">
   233→                    <ShieldCheck className="mr-1 h-3.5 w-3.5" />
   234→                    Bucket <span className="num mx-1">{data.bucketName ?? 'vic'}</span> sẵn sàng
   235→                  </Badge>
   236→                ) : (
   237→                  <Badge className="rounded-full border-rose-500/40 bg-rose-500/10 text-rose-400">
   238→                    <TriangleAlert className="mr-1 h-3.5 w-3.5" />
   239→                    Bucket lỗi: {data.bucketError ?? 'không kết nối được'}
   240→                  </Badge>
   241→                )
   242→              ) : (
   243→                <Skeleton className="h-6 w-44 rounded-full" />
   244→              )}
   245→              <Badge variant="outline" className="rounded-full border-white/10 text-zinc-300">
   246→                <HardDrive className="mr-1 h-3.5 w-3.5" style={{ color: 'var(--accent-2)' }} />
   247→                Dung lượng local: <span className="num ml-1">{formatBytes(totalLocalBytes)}</span>
   248→              </Badge>
   249→              <Badge variant="outline" className="rounded-full border-white/10 text-zinc-300">
   250→                <Cloud className="mr-1 h-3.5 w-3.5" style={{ color: 'var(--accent-1)' }} />
   251→                Dung lượng bucket: <span className="num ml-1">{formatBytes(totalBucketBytes)}</span>
   252→                <span className="num ml-1 text-zinc-500">/ 30GB</span>
   253→              </Badge>
   254→            </div>
   255→          </div>
   256→
   257→          <div className="flex flex-col gap-3 sm:flex-row">
   258→            <Input
   259→              value={name}
   260→              onChange={(e) => setName(e.target.value)}
   261→              onKeyDown={(e) => {
   262→                if (e.key === 'Enter' && !creating) create()
   263→              }}
   264→              placeholder="vd: nexus-full-2025"
   265→              className="h-11 flex-1 rounded-xl border-white/10 bg-white/5 text-zinc-100 placeholder:text-zinc-600"
   266→              aria-label="Tên bản sao lưu"
   267→            />
   268→            <Button
   269→              onClick={create}
   270→              disabled={creating}
   271→              className="h-11 rounded-xl px-6 font-semibold text-black"
   272→              style={{ background: 'var(--accent-1)' }}
   273→            >
   274→              {creating ? (
   275→                <Loader2 className="mr-1.5 h-4 w-4 animate-spin" />
   276→              ) : (
   277→                <DatabaseBackup className="mr-1.5 h-4 w-4" />
   278→              )}
   279→              {creating ? 'Đang tạo backup...' : 'Tạo backup ngay'}
   280→            </Button>
   281→          </div>
   282→        </div>
   283→      </Card>
   284→
   285→      {/* Hai danh sách */}
   286→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-2">
   287→        {/* Local */}
   288→        <Card className="glass rounded-2xl p-4">
   289→          <h3 className="mb-3 flex items-center gap-2 text-sm font-semibold text-zinc-200">
   290→            <HardDrive className="h-4 w-4" style={{ color: 'var(--accent-2)' }} />
   291→            Bản sao lưu trên máy chủ
   292→            <Badge variant="outline" className="num ml-auto rounded-full border-white/10 text-zinc-400">
   293→              {loading ? '...' : local.length}
   294→            </Badge>
   295→          </h3>
   296→          {loading ? (
   297→            <div className="space-y-2">
   298→              <Skeleton className="h-10 w-full rounded-xl" />
   299→              <Skeleton className="h-10 w-full rounded-xl" />
   300→            </div>
   301→          ) : local.length === 0 ? (
   302→            <div className="flex flex-col items-center gap-2 py-10 text-center">
   303→              <PackageOpen className="h-8 w-8 text-zinc-600" />
   304→              <p className="text-sm text-zinc-500">
   305→                Chưa có bản backup nào trên máy chủ — tạo bản đầu tiên ở trên.
   306→              </p>
   307→            </div>
   308→          ) : (
   309→            <div className="thin-scroll max-h-96 overflow-y-auto">
   310→              <Table>
   311→                <TableHeader>
   312→                  <TableRow className="border-white/10">
   313→                    <TableHead className="text-zinc-400">Tên</TableHead>
   314→                    <TableHead className="text-zinc-400">Dung lượng</TableHead>
   315→                    <TableHead className="text-zinc-400">Thời gian</TableHead>
   316→                    <TableHead className="text-zinc-400">Bucket</TableHead>
   317→                    <TableHead className="text-right text-zinc-400">Hành động</TableHead>
   318→                  </TableRow>
   319→                </TableHeader>
   320→                <TableBody>
   321→                  {local.map((b) => (
   322→                    <TableRow key={b.id} className="border-white/5">
   323→                      <TableCell className="max-w-[180px] truncate font-medium text-zinc-200" title={b.name}>
   324→                        {b.name}
   325→                      </TableCell>
   326→                      <TableCell className="num whitespace-nowrap text-zinc-300">
   327→                        {formatBytes(b.sizeBytes)}
   328→                      </TableCell>
   329→                      <TableCell className="num whitespace-nowrap text-zinc-400">
   330→                        {formatTime(b.createdAt)}
   331→                      </TableCell>
   332→                      <TableCell>
   333→                        {b.uploaded ? (
   334→                          <Badge className="rounded-full border-emerald-500/40 bg-emerald-500/10 text-emerald-400">
   335→                            Đã lên
   336→                          </Badge>
   337→                        ) : (
   338→                          <Badge variant="outline" className="rounded-full border-white/10 text-zinc-500">
   339→                            Chưa
   340→                          </Badge>
   341→                        )}
   342→                      </TableCell>
   343→                      <TableCell className="text-right">
   344→                        <div className="flex justify-end gap-1">
   345→                          <Button
   346→                            size="sm"
   347→                            variant="ghost"
   348→                            title={b.uploaded ? 'Đã tải lên bucket' : 'Tải lên bucket'}
   349→                            disabled={b.uploaded || busyLocalId === b.id}
   350→                            className="h-8 rounded-lg px-2 text-zinc-300 hover:text-white disabled:opacity-40"
   351→                            onClick={() => uploadOne(b)}
   352→                          >
   353→                            {busyLocalId === b.id ? (
   354→                              <Loader2 className="h-3.5 w-3.5 animate-spin" />
   355→                            ) : (
   356→                              <UploadCloud className="h-3.5 w-3.5" />
   357→                            )}
   358→                          </Button>
   359→                          <Button
   360→                            size="sm"
   361→                            variant="ghost"
   362→                            title="Tải về (link pre-signed)"
   363→                            disabled={!b.uploaded || busyLocalId === b.id}
   364→                            className="h-8 rounded-lg px-2 text-zinc-300 hover:text-white disabled:opacity-40"
   365→                            onClick={() => downloadLocal(b)}
   366→                          >
   367→                            <Download className="h-3.5 w-3.5" />
   368→                          </Button>
   369→                          <AlertDialog>
   370→                            <AlertDialogTrigger asChild>
   371→                              <Button
   372→                                size="sm"
   373→                                variant="ghost"
   374→                                title="Xóa bản backup này"
   375→                                disabled={busyLocalId === b.id}
   376→                                className="h-8 rounded-lg px-2 text-rose-400 hover:text-rose-300"
   377→                              >
   378→                                <Trash2 className="h-3.5 w-3.5" />
   379→                              </Button>
   380→                            </AlertDialogTrigger>
   381→                            <AlertDialogContent className="glass rounded-2xl border-white/10">
   382→                              <AlertDialogHeader>
   383→                                <AlertDialogTitle className="text-zinc-100">
   384→                                  Xóa bản backup trên máy chủ?
   385→                                </AlertDialogTitle>
   386→                                <AlertDialogDescription className="text-zinc-400">
   387→                                  File <span className="num">{b.name}</span> sẽ bị xóa vĩnh viễn khỏi
   388→                                  máy chủ. Bản copy trên bucket (nếu đã tải lên) vẫn được giữ.
   389→                                </AlertDialogDescription>
   390→                              </AlertDialogHeader>
   391→                              <AlertDialogFooter>
   392→                                <AlertDialogCancel className="rounded-xl border-white/10 bg-transparent text-zinc-200 hover:bg-white/10">
   393→                                  Hủy
   394→                                </AlertDialogCancel>
   395→                                <AlertDialogAction
   396→                                  className="rounded-xl bg-rose-600 text-white hover:bg-rose-500"
   397→                                  onClick={() => deleteLocal(b.id)}
   398→                                >
   399→                                  Xóa
   400→                                </AlertDialogAction>
   401→                              </AlertDialogFooter>
   402→                            </AlertDialogContent>
   403→                          </AlertDialog>
   404→                        </div>
   405→                      </TableCell>
   406→                    </TableRow>
   407→                  ))}
   408→                </TableBody>
   409→              </Table>
   410→            </div>
   411→          )}
   412→        </Card>
   413→
   414→        {/* Bucket */}
   415→        <Card className="glass rounded-2xl p-4">
   416→          <h3 className="mb-3 flex items-center gap-2 text-sm font-semibold text-zinc-200">
   417→            <Cloud className="h-4 w-4" style={{ color: 'var(--accent-1)' }} />
   418→            Trên bucket (30GB)
   419→            <Badge variant="outline" className="num ml-auto rounded-full border-white/10 text-zinc-400">
   420→              {loading ? '...' : bucket.length}
   421→            </Badge>
   422→          </h3>
   423→          {loading ? (
   424→            <div className="space-y-2">
   425→              <Skeleton className="h-10 w-full rounded-xl" />
   426→              <Skeleton className="h-10 w-full rounded-xl" />
   427→            </div>
   428→          ) : bucket.length === 0 ? (
   429→            <div className="flex flex-col items-center gap-2 py-10 text-center">
   430→              <Cloud className="h-8 w-8 text-zinc-600" />
   431→              <p className="text-sm text-zinc-500">
   432→                Chưa có backup nào trên bucket — bấm “Tải lên bucket” ở bảng bên trái.
   433→              </p>
   434→            </div>
   435→          ) : (
   436→            <div className="thin-scroll max-h-96 overflow-y-auto">
   437→              <Table>
   438→                <TableHeader>
   439→                  <TableRow className="border-white/10">
   440→                    <TableHead className="text-zinc-400">Key</TableHead>
   441→                    <TableHead className="text-zinc-400">Dung lượng</TableHead>
   442→                    <TableHead className="text-zinc-400">Ngày</TableHead>
   443→                    <TableHead className="text-right text-zinc-400">Hành động</TableHead>
   444→                  </TableRow>
   445→                </TableHeader>
   446→                <TableBody>
   447→                  {bucket.map((item: BackupBucketItem) => {
   448→                    const busy = busyBucketKey === item.key
   449→                    return (
   450→                      <TableRow key={item.key} className="border-white/5">
   451→                        <TableCell className="max-w-[200px] truncate text-zinc-200" title={item.key}>
   452→                          {item.key}
   453→                        </TableCell>
   454→                        <TableCell className="num whitespace-nowrap text-zinc-300">
   455→                          {formatBytes(item.sizeBytes)}
   456→                        </TableCell>
   457→                        <TableCell className="num whitespace-nowrap text-zinc-400">
   458→                          {formatTime(item.lastModified)}
   459→                        </TableCell>
   460→                        <TableCell className="text-right">
   461→                          <div className="flex justify-end gap-1">
   462→                            <Button
   463→                              size="sm"
   464→                              variant="ghost"
   465→                              title="Tải về (link pre-signed)"
   466→                              disabled={busy}
   467→                              className="h-8 rounded-lg px-2 text-zinc-300 hover:text-white"
   468→                              onClick={() => downloadBucket(item.key)}
   469→                            >
   470→                              {busy ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Download className="h-3.5 w-3.5" />}
   471→                            </Button>
   472→                            <AlertDialog>
   473→                              <AlertDialogTrigger asChild>
   474→                                <Button
   475→                                  size="sm"
   476→                                  variant="ghost"
   477→                                  title="Khôi phục về máy chủ"
   478→                                  disabled={busy}
   479→                                  className="h-8 rounded-lg px-2 text-zinc-300 hover:text-white"
   480→                                >
   481→                                  <RotateCcw className="h-3.5 w-3.5" />
   482→                                </Button>
   483→                              </AlertDialogTrigger>
   484→                              <AlertDialogContent className="glass rounded-2xl border-white/10">
   485→                                <AlertDialogHeader>
   486→                                  <AlertDialogTitle className="text-zinc-100">
   487→                                    Khôi phục từ bucket?
   488→                                  </AlertDialogTitle>
   489→                                  <AlertDialogDescription className="text-zinc-400">
