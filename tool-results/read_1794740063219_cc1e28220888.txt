     1→'use client'
     2→
     3→import { useCallback, useEffect, useState } from 'react'
     4→import {
     5→  Cloud,
     6→  DatabaseBackup,
     7→  Download,
     8→  Github,
     9→  HardDrive,
    10→  Loader2,
    11→  PackageOpen,
    12→  RotateCcw,
    13→  ShieldCheck,
    14→  Trash2,
    15→  TriangleAlert,
    16→  UploadCloud,
    17→} from 'lucide-react'
    18→import { Card } from '@/components/ui/card'
    19→import { Button } from '@/components/ui/button'
    20→import { Badge } from '@/components/ui/badge'
    21→import { Input } from '@/components/ui/input'
    22→import { Skeleton } from '@/components/ui/skeleton'
    23→import {
    24→  Table,
    25→  TableBody,
    26→  TableCell,
    27→  TableHead,
    28→  TableHeader,
    29→  TableRow,
    30→} from '@/components/ui/table'
    31→import {
    32→  AlertDialog,
    33→  AlertDialogAction,
    34→  AlertDialogCancel,
    35→  AlertDialogContent,
    36→  AlertDialogDescription,
    37→  AlertDialogFooter,
    38→  AlertDialogHeader,
    39→  AlertDialogTitle,
    40→  AlertDialogTrigger,
    41→} from '@/components/ui/alert-dialog'
    42→import { toast } from 'sonner'
    43→import type { BackupBucketItem, BackupListResponse, BackupLocal } from '@/lib/types'
    44→import { formatBytes, formatTime } from '@/lib/format'
    45→
    46→type ListData = BackupListResponse & { totalLocalBytes?: number }
    47→
    48→export function BackupPanel() {
    49→  const [data, setData] = useState<ListData | null>(null)
    50→  const [loading, setLoading] = useState(true)
    51→  const [name, setName] = useState('')
    52→  const [creating, setCreating] = useState(false)
    53→  const [busyLocalId, setBusyLocalId] = useState<number | null>(null)
    54→  const [busyBucketKey, setBusyBucketKey] = useState<string | null>(null)
    55→
    56→  const local = data?.local ?? []
    57→  const bucket = data?.bucket ?? []
    58→  const totalLocalBytes = data?.totalLocalBytes ?? local.reduce((s, b) => s + b.sizeBytes, 0)
    59→  const totalBucketBytes = bucket.reduce((s, b) => s + b.sizeBytes, 0)
    60→
    61→  const load = useCallback(async () => {
    62→    try {
    63→      const r = await fetch('/api/backups', { cache: 'no-store' })
    64→      const json = (await r.json().catch(() => null)) as (ListData & { error?: string }) | null
    65→      if (!r.ok || !json || json.error) throw new Error(json?.error ?? 'Không tải được danh sách backup')
    66→      setData(json)
    67→    } catch (e) {
    68→      /* giữ danh sách cũ khi lỗi — không làm trống bảng */
    69→      toast.error((e as Error).message || 'Không tải được danh sách backup')
    70→    } finally {
    71→      setLoading(false)
    72→    }
    73→  }, [])
    74→
    75→  useEffect(() => {
    76→    load()
    77→  }, [load])
    78→
    79→  const create = async () => {
    80→    setCreating(true)
    81→    try {
    82→      const r = await fetch('/api/backups', {
    83→        method: 'POST',
    84→        headers: { 'Content-Type': 'application/json' },
    85→        body: JSON.stringify({ name: name.trim() || 'backup' }),
    86→      })
    87→      const json = (await r.json()) as { backup?: BackupLocal; error?: string }
    88→      if (!r.ok || json.error) throw new Error(json.error || 'Tạo backup thất bại')
    89→      toast.success(
    90→        `Đã tạo backup ${json.backup?.name ?? ''} (${formatBytes(json.backup?.sizeBytes ?? 0)})`,
    91→      )
    92→      setName('')
    93→      await load()
    94→    } catch (e) {
    95→      toast.error((e as Error).message || 'Tạo backup thất bại')
    96→    } finally {
    97→      setCreating(false)
    98→    }
    99→  }
   100→
   101→  const uploadOne = async (b: BackupLocal) => {
   102→    setBusyLocalId(b.id)
   103→    toast.info(`Đang tải ${b.name} lên bucket...`)
   104→    try {
   105→      const r = await fetch('/api/backups', {
   106→        method: 'PUT',
   107→        headers: { 'Content-Type': 'application/json' },
   108→        body: JSON.stringify({ action: 'upload', id: b.id }),
   109→      })
   110→      const json = (await r.json()) as { ok?: boolean; error?: string; key?: string }
   111→      if (!r.ok || !json.ok) throw new Error(json.error || 'Tải lên bucket thất bại')
   112→      toast.success(`Đã tải lên bucket: ${json.key ?? b.name}`)
   113→      await load()
   114→    } catch (e) {
   115→      toast.error((e as Error).message || 'Tải lên bucket thất bại')
   116→    } finally {
   117→      setBusyLocalId(null)
   118→    }
   119→  }
   120→
   121→  const downloadLocal = async (b: BackupLocal) => {
   122→    setBusyLocalId(b.id)
   123→    try {
   124→      const r = await fetch('/api/backups', {
   125→        method: 'PUT',
   126→        headers: { 'Content-Type': 'application/json' },
   127→        body: JSON.stringify({ action: 'presign', id: b.id }),
   128→      })
   129→      const json = (await r.json()) as { url?: string; error?: string }
   130→      if (!r.ok || !json.url) throw new Error(json.error || 'Không tạo được link tải về')
   131→      window.open(json.url, '_blank', 'noopener')
   132→      toast.success('Đã mở link tải về (hết hạn sau 1 giờ)')
   133→    } catch (e) {
   134→      toast.error((e as Error).message || 'Không tạo được link tải về')
   135→    } finally {
   136→      setBusyLocalId(null)
   137→    }
   138→  }
   139→
   140→  const downloadBucket = async (key: string) => {
   141→    setBusyBucketKey(key)
   142→    try {
   143→      const r = await fetch('/api/backups', {
   144→        method: 'PUT',
   145→        headers: { 'Content-Type': 'application/json' },
   146→        body: JSON.stringify({ action: 'presign', key }),
   147→      })
   148→      const json = (await r.json()) as { url?: string; error?: string }
   149→      if (!r.ok || !json.url) throw new Error(json.error || 'Không tạo được link tải về')
   150→      window.open(json.url, '_blank', 'noopener')
   151→      toast.success('Đã mở link tải về (hết hạn sau 1 giờ)')
   152→    } catch (e) {
   153→      toast.error((e as Error).message || 'Không tạo được link tải về')
   154→    } finally {
   155→      setBusyBucketKey(null)
   156→    }
   157→  }
   158→
   159→  const restoreBucket = async (key: string) => {
   160→    setBusyBucketKey(key)
   161→    toast.info('Đang khôi phục backup về máy chủ...')
   162→    try {
   163→      const r = await fetch('/api/backups', {
   164→        method: 'PUT',
   165→        headers: { 'Content-Type': 'application/json' },
   166→        body: JSON.stringify({ action: 'restore', key }),
   167→      })
   168→      const json = (await r.json()) as { ok?: boolean; message?: string; error?: string }
   169→      if (!r.ok || !json.ok) throw new Error(json.error || 'Khôi phục thất bại')
   170→      toast.success(json.message || 'Đã khôi phục backup về máy chủ')
   171→    } catch (e) {
   172→      toast.error((e as Error).message || 'Khôi phục thất bại')
   173→    } finally {
   174→      setBusyBucketKey(null)
   175→    }
   176→  }
   177→
   178→  const deleteLocal = async (id: number) => {
   179→    setBusyLocalId(id)
   180→    try {
   181→      const r = await fetch(`/api/backups?type=local&id=${id}`, { method: 'DELETE' })
   182→      const json = (await r.json()) as { success?: boolean; error?: string }
   183→      if (!r.ok || !json.success) throw new Error(json.error || 'Không xóa được bản backup')
   184→      toast.success('Đã xóa bản backup trên máy chủ')
   185→      await load()
   186→    } catch (e) {
   187→      toast.error((e as Error).message || 'Không xóa được bản backup')
   188→    } finally {
   189→      setBusyLocalId(null)
   190→    }
   191→  }
   192→
   193→  const deleteBucket = async (key: string) => {
   194→    setBusyBucketKey(key)
   195→    try {
   196→      const r = await fetch(`/api/backups?type=bucket&key=${encodeURIComponent(key)}`, {
   197→        method: 'DELETE',
   198→      })
   199→      const json = (await r.json()) as { success?: boolean; error?: string }
   200→      if (!r.ok || !json.success) throw new Error(json.error || 'Không xóa được object trên bucket')
   201→      toast.success('Đã xóa object trên bucket')
   202→      await load()
   203→    } catch (e) {
   204→      toast.error((e as Error).message || 'Không xóa được object trên bucket')
   205→    } finally {
   206→      setBusyBucketKey(null)
   207→    }
   208→  }
   209→
   210→  return (
   211→    <div className="space-y-4">
   212→      {/* Hero: tạo backup + trạng thái */}
   213→      <Card className="glass rounded-2xl p-6">
   214→        <div className="flex flex-col gap-5">
   215→          <div className="flex flex-wrap items-center justify-between gap-4">
   216→            <div className="flex items-center gap-3">
   217→              <div
   218→                className="flex h-11 w-11 items-center justify-center rounded-xl border border-white/10"
   219→                style={{ background: 'color-mix(in srgb, var(--accent-1) 14%, transparent)' }}
   220→              >
   221→                <DatabaseBackup className="h-5 w-5" style={{ color: 'var(--accent-1)' }} />
   222→              </div>
   223→              <div>
   224→                <h3 className="text-base font-semibold text-zinc-100">Tạo bản sao lưu</h3>
   225→                <p className="text-xs text-zinc-500">
   226→                  Nén toàn bộ mã nguồn + cấu hình thành file .tar.gz (loại trừ node_modules,
   227→                  .next, upload, .env)
   228→                </p>
   229→              </div>
   230→            </div>
   231→            <div className="flex flex-wrap items-center gap-2">
   232→              {data ? (
   233→                data.bucketOk ? (
   234→                  <Badge className="rounded-full border-emerald-500/40 bg-emerald-500/10 text-emerald-400">
   235→                    <ShieldCheck className="mr-1 h-3.5 w-3.5" />
   236→                    Bucket <span className="num mx-1">{data.bucketName ?? 'vic'}</span> sẵn sàng
   237→                  </Badge>
   238→                ) : (
   239→                  <Badge className="rounded-full border-rose-500/40 bg-rose-500/10 text-rose-400">
   240→                    <TriangleAlert className="mr-1 h-3.5 w-3.5" />
   241→                    Bucket lỗi: {data.bucketError ?? 'không kết nối được'}
   242→                  </Badge>
   243→                )
   244→              ) : (
   245→                <Skeleton className="h-6 w-44 rounded-full" />
   246→              )}
   247→              <Badge variant="outline" className="rounded-full border-white/10 text-zinc-300">
   248→                <HardDrive className="mr-1 h-3.5 w-3.5" style={{ color: 'var(--accent-2)' }} />
   249→                Dung lượng local: <span className="num ml-1">{formatBytes(totalLocalBytes)}</span>
   250→              </Badge>
   251→              <Badge variant="outline" className="rounded-full border-white/10 text-zinc-300">
   252→                <Cloud className="mr-1 h-3.5 w-3.5" style={{ color: 'var(--accent-1)' }} />
   253→                Dung lượng bucket: <span className="num ml-1">{formatBytes(totalBucketBytes)}</span>
   254→                <span className="num ml-1 text-zinc-500">/ 30GB</span>
   255→              </Badge>
   256→            </div>
   257→          </div>
   258→
   259→          <div className="flex flex-col gap-3 sm:flex-row">
   260→            <Input
   261→              value={name}
   262→              onChange={(e) => setName(e.target.value)}
   263→              onKeyDown={(e) => {
   264→                if (e.key === 'Enter' && !creating) create()
   265→              }}
   266→              placeholder="vd: nexus-full-2025"
   267→              className="h-11 flex-1 rounded-xl border-white/10 bg-white/5 text-zinc-100 placeholder:text-zinc-400"
   268→              aria-label="Tên bản sao lưu"
   269→            />
   270→            <Button
   271→              onClick={create}
   272→              disabled={creating}
   273→              className="h-11 rounded-xl px-6 font-semibold text-black"
   274→              style={{ background: 'var(--accent-1)' }}
   275→            >
   276→              {creating ? (
   277→                <Loader2 className="mr-1.5 h-4 w-4 animate-spin" />
   278→              ) : (
   279→                <DatabaseBackup className="mr-1.5 h-4 w-4" />
   280→              )}
   281→              {creating ? 'Đang tạo backup...' : 'Tạo backup ngay'}
   282→            </Button>
   283→          </div>
   284→        </div>
   285→      </Card>
   286→
   287→      {/* Hai danh sách */}
   288→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-2">
   289→        {/* Local */}
   290→        <Card className="glass rounded-2xl p-4">
   291→          <h3 className="mb-3 flex items-center gap-2 text-sm font-semibold text-zinc-200">
   292→            <HardDrive className="h-4 w-4" style={{ color: 'var(--accent-2)' }} />
   293→            Bản sao lưu trên máy chủ
   294→            <Badge variant="outline" className="num ml-auto rounded-full border-white/10 text-zinc-400">
   295→              {loading ? '...' : local.length}
   296→            </Badge>
   297→          </h3>
   298→          {loading ? (
   299→            <div className="space-y-2">
   300→              <Skeleton className="h-10 w-full rounded-xl" />
   301→              <Skeleton className="h-10 w-full rounded-xl" />
   302→            </div>
   303→          ) : local.length === 0 ? (
   304→            <div className="flex flex-col items-center gap-2 py-10 text-center">
   305→              <PackageOpen className="h-8 w-8 text-zinc-400" />
   306→              <p className="text-sm text-zinc-500">
   307→                Chưa có bản backup nào trên máy chủ — tạo bản đầu tiên ở trên.
   308→              </p>
   309→            </div>
   310→          ) : (
   311→            <div className="thin-scroll max-h-96 overflow-y-auto">
   312→              <Table>
   313→                <TableHeader>
   314→                  <TableRow className="border-white/10">
   315→                    <TableHead className="text-zinc-400">Tên</TableHead>
   316→                    <TableHead className="text-zinc-400">Dung lượng</TableHead>
   317→                    <TableHead className="text-zinc-400">Thời gian</TableHead>
   318→                    <TableHead className="text-zinc-400">Bucket</TableHead>
   319→                    <TableHead className="text-right text-zinc-400">Hành động</TableHead>
   320→                  </TableRow>
   321→                </TableHeader>
   322→                <TableBody>
   323→                  {local.map((b) => (
   324→                    <TableRow key={b.id} className="border-white/5">
   325→                      <TableCell className="max-w-[180px] truncate font-medium text-zinc-200" title={b.name}>
   326→                        {b.name}
   327→                      </TableCell>
   328→                      <TableCell className="num whitespace-nowrap text-zinc-300">
   329→                        {formatBytes(b.sizeBytes)}
   330→                      </TableCell>
   331→                      <TableCell className="num whitespace-nowrap text-zinc-400">
   332→                        {formatTime(b.createdAt)}
   333→                      </TableCell>
   334→                      <TableCell>
   335→                        {b.uploaded ? (
   336→                          <Badge className="rounded-full border-emerald-500/40 bg-emerald-500/10 text-emerald-400">
   337→                            Đã tải lên
   338→                          </Badge>
   339→                        ) : (
   340→                          <Badge variant="outline" className="rounded-full border-white/10 text-zinc-500">
   341→                            Chưa tải lên
   342→                          </Badge>
   343→                        )}
   344→                      </TableCell>
   345→                      <TableCell className="text-right">
   346→                        <div className="flex justify-end gap-1">
   347→                          <Button
   348→                            size="sm"
   349→                            variant="ghost"
   350→                            title={b.uploaded ? 'Đã tải lên bucket' : 'Tải lên bucket'}
   351→                            aria-label={b.uploaded ? `Bản backup ${b.name} đã tải lên bucket` : `Tải bản backup ${b.name} lên bucket`}
   352→                            disabled={b.uploaded || busyLocalId === b.id}
   353→                            className="h-9 rounded-lg px-2.5 text-zinc-300 hover:text-white disabled:opacity-40"
   354→                            onClick={() => uploadOne(b)}
   355→                          >
   356→                            {busyLocalId === b.id ? (
   357→                              <Loader2 className="h-3.5 w-3.5 animate-spin" />
   358→                            ) : (
   359→                              <UploadCloud className="h-3.5 w-3.5" />
   360→                            )}
   361→                          </Button>
   362→                          <Button
   363→                            size="sm"
   364→                            variant="ghost"
   365→                            title="Tải về (link pre-signed)"
   366→                            aria-label={`Tải về bản backup ${b.name}`}
   367→                            disabled={!b.uploaded || busyLocalId === b.id}
   368→                            className="h-9 rounded-lg px-2.5 text-zinc-300 hover:text-white disabled:opacity-40"
   369→                            onClick={() => downloadLocal(b)}
   370→                          >
   371→                            <Download className="h-3.5 w-3.5" />
   372→                          </Button>
   373→                          <AlertDialog>
   374→                            <AlertDialogTrigger asChild>
   375→                              <Button
   376→                                size="sm"
   377→                                variant="ghost"
   378→                                title="Xóa bản backup này"
   379→                                aria-label={`Xóa bản backup ${b.name}`}
   380→                                disabled={busyLocalId === b.id}
   381→                                className="h-9 rounded-lg px-2.5 text-rose-400 hover:text-rose-300"
   382→                              >
   383→                                <Trash2 className="h-3.5 w-3.5" />
   384→                              </Button>
   385→                            </AlertDialogTrigger>
   386→                            <AlertDialogContent className="glass rounded-2xl border-white/10">
   387→                              <AlertDialogHeader>
   388→                                <AlertDialogTitle className="text-zinc-100">
   389→                                  Xóa bản backup trên máy chủ?
   390→                                </AlertDialogTitle>
   391→                                <AlertDialogDescription className="text-zinc-400">
   392→                                  File <span className="num">{b.name}</span> sẽ bị xóa vĩnh viễn khỏi
   393→                                  máy chủ. Bản copy trên bucket (nếu đã tải lên) vẫn được giữ.
   394→                                </AlertDialogDescription>
   395→                              </AlertDialogHeader>
   396→                              <AlertDialogFooter>
   397→                                <AlertDialogCancel className="rounded-xl border-white/10 bg-transparent text-zinc-200 hover:bg-white/10">
   398→                                  Hủy
   399→                                </AlertDialogCancel>
   400→                                <AlertDialogAction
   401→                                  className="rounded-xl bg-rose-600 text-white hover:bg-rose-500"
   402→                                  onClick={() => deleteLocal(b.id)}
   403→                                >
   404→                                  Xóa
   405→                                </AlertDialogAction>
   406→                              </AlertDialogFooter>
   407→                            </AlertDialogContent>
   408→                          </AlertDialog>
   409→                        </div>
   410→                      </TableCell>
   411→                    </TableRow>
   412→                  ))}
   413→                </TableBody>
   414→              </Table>
   415→            </div>
   416→          )}
   417→        </Card>
   418→
   419→        {/* Bucket */}
   420→        <Card className="glass rounded-2xl p-4">
   421→          <h3 className="mb-3 flex items-center gap-2 text-sm font-semibold text-zinc-200">
   422→            <Cloud className="h-4 w-4" style={{ color: 'var(--accent-1)' }} />
   423→            Trên bucket (30GB)
   424→            <Badge variant="outline" className="num ml-auto rounded-full border-white/10 text-zinc-400">
   425→              {loading ? '...' : bucket.length}
   426→            </Badge>
   427→          </h3>
   428→          {loading ? (
   429→            <div className="space-y-2">
   430→              <Skeleton className="h-10 w-full rounded-xl" />
   431→              <Skeleton className="h-10 w-full rounded-xl" />
   432→            </div>
   433→          ) : bucket.length === 0 ? (
   434→            <div className="flex flex-col items-center gap-2 py-10 text-center">
   435→              <Cloud className="h-8 w-8 text-zinc-400" />
   436→              <p className="text-sm text-zinc-500">
   437→                Chưa có backup nào trên bucket — bấm “Tải lên bucket” ở bảng bên trái.
   438→              </p>
   439→            </div>
   440→          ) : (
   441→            <div className="thin-scroll max-h-96 overflow-y-auto">
   442→              <Table>
   443→                <TableHeader>
   444→                  <TableRow className="border-white/10">
   445→                    <TableHead className="text-zinc-400">Key</TableHead>
   446→                    <TableHead className="text-zinc-400">Dung lượng</TableHead>
   447→                    <TableHead className="text-zinc-400">Ngày</TableHead>
   448→                    <TableHead className="text-right text-zinc-400">Hành động</TableHead>
   449→                  </TableRow>
   450→                </TableHeader>
   451→                <TableBody>
   452→                  {bucket.map((item: BackupBucketItem) => {
   453→                    const busy = busyBucketKey === item.key
   454→                    return (
   455→                      <TableRow key={item.key} className="border-white/5">
   456→                        <TableCell className="max-w-[200px] truncate text-zinc-200" title={item.key}>
   457→                          {item.key}
   458→                        </TableCell>
   459→                        <TableCell className="num whitespace-nowrap text-zinc-300">
   460→                          {formatBytes(item.sizeBytes)}
   461→                        </TableCell>
   462→                        <TableCell className="num whitespace-nowrap text-zinc-400">
   463→                          {formatTime(item.lastModified)}
   464→                        </TableCell>
   465→                        <TableCell className="text-right">
   466→                          <div className="flex justify-end gap-1">
   467→                            <Button
   468→                              size="sm"
   469→                              variant="ghost"
   470→                              title="Tải về (link pre-signed)"
   471→                              aria-label={`Tải về backup ${item.key} từ bucket`}
   472→                              disabled={busy}
   473→                              className="h-9 rounded-lg px-2.5 text-zinc-300 hover:text-white"
   474→                              onClick={() => downloadBucket(item.key)}
   475→                            >
   476→                              {busy ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Download className="h-3.5 w-3.5" />}
   477→                            </Button>
   478→                            <AlertDialog>
   479→                              <AlertDialogTrigger asChild>
   480→                                <Button
   481→                                  size="sm"
