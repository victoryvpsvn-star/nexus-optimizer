    21→    21→import { Input } from '@/components/ui/input'

    30→    30→} from '@/components/ui/table'

    39→    39→  AlertDialogTitle,
    40→    40→  AlertDialogTrigger,
    41→    41→} from '@/components/ui/alert-dialog'
    42→    42→import { toast } from 'sonner'
    43→    43→import type { BackupBucketItem, BackupListResponse, BackupLocal } from '@/lib/types'
    44→    44→import { formatBytes, formatTime } from '@/lib/format'
    45→    45→
    46→    46→type ListData = BackupListResponse & { totalLocalBytes?: number }
    47→    47→
    48→    48→export function BackupPanel() {
    49→    49→  const [data, setData] = useState<ListData | null>(null)
    50→    50→  const [loading, setLoading] = useState(true)
    51→    51→  const [name, setName] = useState('')
    52→    52→  const [creating, setCreating] = useState(false)
    53→    53→  const [busyLocalId, setBusyLocalId] = useState<number | null>(null)
    54→    54→  const [busyBucketKey, setBusyBucketKey] = useState<string | null>(null)
    55→    55→
    56→    56→  const local = data?.local ?? []
    57→    57→  const bucket = data?.bucket ?? []
    58→    58→  const totalLocalBytes = data?.totalLocalBytes ?? local.reduce((s, b) => s + b.sizeBytes, 0)
    59→    59→  const totalBucketBytes = bucket.reduce((s, b) => s + b.sizeBytes, 0)
    60→    60→
    61→    61→  const load = useCallback(async () => {
    62→    62→    try {
    63→    63→      const r = await fetch('/api/backups', { cache: 'no-store' })
    64→    64→      const json = (await r.json()) as ListData
    65→    65→      setData(json)
    66→    66→    } catch {
    67→    67→      toast.error('Không tải được danh sách bản sao lưu')
    68→    68→    } finally {
    69→    69→      setLoading(false)
    70→    70→    }
    71→    71→  }, [])
    72→    72→
    73→    73→  useEffect(() => {
    74→    74→    load()
    75→    75→  }, [load])
    76→    76→
    77→    77→  const create = async () => {
    78→    78→    setCreating(true)
    79→    79→    try {
    80→    80→      const r = await fetch('/api/backups', {
    81→    81→        method: 'POST',
    82→    82→        headers: { 'Content-Type': 'application/json' },
    83→    83→        body: JSON.stringify({ name: name.trim() || 'backup' }),
    84→    84→      })
    85→    85→      const json = (await r.json()) as { backup?: BackupLocal; error?: string }
    86→    86→      if (!r.ok || json.error) throw new Error(json.error || 'Tạo backup thất bại')
    87→    87→      toast.success(
    88→    88→        `Đã tạo backup ${json.backup?.name ?? ''} (${formatBytes(json.backup?.sizeBytes ?? 0)})`,
    89→    89→      )
    90→    90→      setName('')
    91→    91→      await load()
    92→    92→    } catch (e) {
    93→    93→      toast.error((e as Error).message || 'Tạo backup thất bại')
    94→    94→    } finally {
    95→    95→      setCreating(false)
    96→    96→    }
    97→    97→  }
    98→    98→
    99→    99→  const uploadOne = async (b: BackupLocal) => {
   100→   100→    setBusyLocalId(b.id)
   101→   101→    toast.info(`Đang tải ${b.name} lên bucket...`)
   102→   102→    try {
   103→   103→      const r = await fetch('/api/backups', {
   104→   104→        method: 'PUT',
   105→   105→        headers: { 'Content-Type': 'application/json' },
   106→   106→        body: JSON.stringify({ action: 'upload', id: b.id }),
   107→   107→      })
   108→   108→      const json = (await r.json()) as { ok?: boolean; error?: string; key?: string }
   109→   109→      if (!r.ok || !json.ok) throw new Error(json.error || 'Tải lên bucket thất bại')
   110→   110→      toast.success(`Đã tải lên bucket: ${json.key ?? b.name}`)
   111→   111→      await load()
   112→   112→    } catch (e) {
   113→   113→      toast.error((e as Error).message || 'Tải lên bucket thất bại')
   114→   114→    } finally {
   115→   115→      setBusyLocalId(null)
   116→   116→    }
   117→   117→  }
   118→   118→
   119→   119→  const downloadLocal = async (b: BackupLocal) => {
   120→   120→    setBusyLocalId(b.id)
   121→   121→    try {
   122→   122→      const r = await fetch('/api/backups', {
   123→   123→        method: 'PUT',
   124→   124→        headers: { 'Content-Type': 'application/json' },
   125→   125→        body: JSON.stringify({ action: 'presign', id: b.id }),
   126→   126→      })
   127→   127→      const json = (await r.json()) as { url?: string; error?: string }
   128→   128→      if (!r.ok || !json.url) throw new Error(json.error || 'Không tạo được link tải về')
   129→   129→      window.open(json.url, '_blank', 'noopener')
   130→   130→      toast.success('Đã mở link tải về (hết hạn sau 1 giờ)')
   131→   131→    } catch (e) {
   132→   132→      toast.error((e as Error).message || 'Không tạo được link tải về')
   133→   133→    } finally {
   134→   134→      setBusyLocalId(null)
   135→   135→    }
   136→   136→  }
   137→   137→
   138→   138→  const downloadBucket = async (key: string) => {
   139→   139→    setBusyBucketKey(key)
   140→   140→    try {
   141→   141→      const r = await fetch('/api/backups', {
   142→   142→        method: 'PUT',
   143→   143→        headers: { 'Content-Type': 'application/json' },
   144→   144→        body: JSON.stringify({ action: 'presign', key }),
   145→   145→      })
   146→   146→      const json = (await r.json()) as { url?: string; error?: string }
   147→   147→      if (!r.ok || !json.url) throw new Error(json.error || 'Không tạo được link tải về')
   148→   148→      window.open(json.url, '_blank', 'noopener')
   149→   149→      toast.success('Đã mở link tải về (hết hạn sau 1 giờ)')
   150→   150→    } catch (e) {
   151→   151→      toast.error((e as Error).message || 'Không tạo được link tải về')
   152→   152→    } finally {
   153→   153→      setBusyBucketKey(null)
   154→   154→    }
   155→   155→  }
   156→   156→
   157→   157→  const restoreBucket = async (key: string) => {
   158→   158→    setBusyBucketKey(key)
   159→   159→    toast.info('Đang khôi phục backup về máy chủ...')
   160→   160→    try {
   161→   161→      const r = await fetch('/api/backups', {
   162→   162→        method: 'PUT',
   163→   163→        headers: { 'Content-Type': 'application/json' },
   164→   164→        body: JSON.stringify({ action: 'restore', key }),
   165→   165→      })
   166→   166→      const json = (await r.json()) as { ok?: boolean; message?: string; error?: string }
   167→   167→      if (!r.ok || !json.ok) throw new Error(json.error || 'Khôi phục thất bại')
   168→   168→      toast.success(json.message || 'Đã khôi phục backup về máy chủ')
   169→   169→    } catch (e) {
   170→   170→      toast.error((e as Error).message || 'Khôi phục thất bại')
   171→   171→    } finally {
   172→   172→      setBusyBucketKey(null)
   173→   173→    }
   174→   174→  }
   175→   175→
   176→   176→  const deleteLocal = async (id: number) => {
   177→   177→    setBusyLocalId(id)
   178→   178→    try {
   179→   179→      const r = await fetch(`/api/backups?type=local&id=${id}`, { method: 'DELETE' })
   180→   180→      const json = (await r.json()) as { success?: boolean; error?: string }
   181→   181→      if (!r.ok || !json.success) throw new Error(json.error || 'Không xóa được bản backup')
   182→   182→      toast.success('Đã xóa bản backup trên máy chủ')
   183→   183→      await load()
   184→   184→    } catch (e) {
   185→   185→      toast.error((e as Error).message || 'Không xóa được bản backup')
   186→   186→    } finally {
   187→   187→      setBusyLocalId(null)
   188→   188→    }
   189→   189→  }
   190→   190→
   191→   191→  const deleteBucket = async (key: string) => {
   192→   192→    setBusyBucketKey(key)
   193→   193→    try {
   194→   194→      const r = await fetch(`/api/backups?type=bucket&key=${encodeURIComponent(key)}`, {
   195→   195→        method: 'DELETE',
   196→   196→      })
   197→   197→      const json = (await r.json()) as { success?: boolean; error?: string }
   198→   198→      if (!r.ok || !json.success) throw new Error(json.error || 'Không xóa được object trên bucket')
   199→   199→      toast.success('Đã xóa object trên bucket')
   200→   200→      await load()
   201→   201→    } catch (e) {
   202→   202→      toast.error((e as Error).message || 'Không xóa được object trên bucket')
   203→   203→    } finally {
   204→   204→      setBusyBucketKey(null)
   205→   205→    }
   206→   206→  }
   207→   207→
   208→   208→  return (
   209→   209→    <div className="space-y-4">
   210→   210→      {/* Hero: tạo backup + trạng thái */}
   211→   211→      <Card className="glass rounded-2xl p-6">
   212→   212→        <div className="flex flex-col gap-5">
   213→   213→          <div className="flex flex-wrap items-center justify-between gap-4">
   214→   214→            <div className="flex items-center gap-3">
   215→   215→              <div
   216→   216→                className="flex h-11 w-11 items-center justify-center rounded-xl border border-white/10"
   217→   217→                style={{ background: 'color-mix(in srgb, var(--accent-1) 14%, transparent)' }}
   218→   218→              >
   219→   219→                <DatabaseBackup className="h-5 w-5" style={{ color: 'var(--accent-1)' }} />
   220→   220→              </div>
   221→   221→              <div>
   222→   222→                <h3 className="text-base font-semibold text-zinc-100">Tạo bản sao lưu</h3>
   223→   223→                <p className="text-xs text-zinc-500">
   224→   224→                  Nén toàn bộ mã nguồn + cấu hình thành file .tar.gz (loại trừ node_modules,
   225→   225→                  .next, upload, .env)
   226→   226→                </p>
   227→   227→              </div>
   228→   228→            </div>
   229→   229→            <div className="flex flex-wrap items-center gap-2">
   230→   230→              {data ? (
   231→   231→                data.bucketOk ? (
   232→   232→                  <Badge className="rounded-full border-emerald-500/40 bg-emerald-500/10 text-emerald-400">
   233→   233→                    <ShieldCheck className="mr-1 h-3.5 w-3.5" />
   234→   234→                    Bucket <span className="num mx-1">{data.bucketName ?? 'vic'}</span> sẵn sàng
   235→   235→                  </Badge>
   236→   236→                ) : (
   237→   237→                  <Badge className="rounded-full border-rose-500/40 bg-rose-500/10 text-rose-400">
   238→   238→                    <TriangleAlert className="mr-1 h-3.5 w-3.5" />
   239→   239→                    Bucket lỗi: {data.bucketError ?? 'không kết nối được'}
   240→   240→                  </Badge>
   241→   241→                )
   242→   242→              ) : (
   243→   243→                <Skeleton className="h-6 w-44 rounded-full" />
   244→   244→              )}
   245→   245→              <Badge variant="outline" className="rounded-full border-white/10 text-zinc-300">
   246→   246→                <HardDrive className="mr-1 h-3.5 w-3.5" style={{ color: 'var(--accent-2)' }} />
   247→   247→                Dung lượng local: <span className="num ml-1">{formatBytes(totalLocalBytes)}</span>
   248→   248→              </Badge>
   249→   249→              <Badge variant="outline" className="rounded-full border-white/10 text-zinc-300">
   250→   250→                <Cloud className="mr-1 h-3.5 w-3.5" style={{ color: 'var(--accent-1)' }} />
   251→   251→                Dung lượng bucket: <span className="num ml-1">{formatBytes(totalBucketBytes)}</span>
   252→   252→                <span className="num ml-1 text-zinc-500">/ 30GB</span>
   253→   253→              </Badge>
   254→   254→            </div>
   255→   255→          </div>
   256→   256→
   257→   257→          <div className="flex flex-col gap-3 sm:flex-row">
   258→   258→            <Input
   259→   259→              value={name}
   260→   260→              onChange={(e) => setName(e.target.value)}
   261→   261→              onKeyDown={(e) => {
   262→   262→                if (e.key === 'Enter' && !creating) create()
   263→   263→              }}
   264→   264→              placeholder="vd: nexus-full-2025"
   265→   265→              className="h-11 flex-1 rounded-xl border-white/10 bg-white/5 text-zinc-100 placeholder:text-zinc-600"
   266→   266→              aria-label="Tên bản sao lưu"
   267→   267→            />
   268→   268→            <Button
   269→   269→              onClick={create}
   270→   270→              disabled={creating}
   271→   271→              className="h-11 rounded-xl px-6 font-semibold text-black"
   272→   272→              style={{ background: 'var(--accent-1)' }}
   273→   273→            >
   274→   274→              {creating ? (
   275→   275→                <Loader2 className="mr-1.5 h-4 w-4 animate-spin" />
   276→   276→              ) : (
   277→   277→                <DatabaseBackup className="mr-1.5 h-4 w-4" />
   278→   278→              )}
   279→   279→              {creating ? 'Đang tạo backup...' : 'Tạo backup ngay'}
   280→   280→            </Button>
   281→   281→          </div>
   282→   282→        </div>
   283→   283→      </Card>
   284→   284→
   285→   285→      {/* Hai danh sách */}
   286→   286→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-2">
   287→   287→        {/* Local */}
   288→   288→        <Card className="glass rounded-2xl p-4">
   289→   289→          <h3 className="mb-3 flex items-center gap-2 text-sm font-semibold text-zinc-200">
   290→   290→            <HardDrive className="h-4 w-4" style={{ color: 'var(--accent-2)' }} />
   291→   291→            Bản sao lưu trên máy chủ
   292→   292→            <Badge variant="outline" className="num ml-auto rounded-full border-white/10 text-zinc-400">
   293→   293→              {loading ? '...' : local.length}
   294→   294→            </Badge>
   295→   295→          </h3>
   296→   296→          {loading ? (
   297→   297→            <div className="space-y-2">
   298→   298→              <Skeleton className="h-10 w-full rounded-xl" />
   299→   299→              <Skeleton className="h-10 w-full rounded-xl" />
   300→   300→            </div>
   301→   301→          ) : local.length === 0 ? (
   302→   302→            <div className="flex flex-col items-center gap-2 py-10 text-center">
   303→   303→              <PackageOpen className="h-8 w-8 text-zinc-600" />
   304→   304→              <p className="text-sm text-zinc-500">
   305→   305→                Chưa có bản backup nào trên máy chủ — tạo bản đầu tiên ở trên.
   306→   306→              </p>
   307→   307→            </div>
   308→   308→          ) : (
   309→   309→            <div className="thin-scroll max-h-96 overflow-y-auto">
   310→   310→              <Table>
   311→   311→                <TableHeader>
   312→   312→                  <TableRow className="border-white/10">
   313→   313→                    <TableHead className="text-zinc-400">Tên</TableHead>
   314→   314→                    <TableHead className="text-zinc-400">Dung lượng</TableHead>
   315→   315→                    <TableHead className="text-zinc-400">Thời gian</TableHead>
   316→   316→                    <TableHead className="text-zinc-400">Bucket</TableHead>
   317→   317→                    <TableHead className="text-right text-zinc-400">Hành động</TableHead>
   318→   318→                  </TableRow>
   319→   319→                </TableHeader>
   320→   320→                <TableBody>
   321→   321→                  {local.map((b) => (
   322→   322→                    <TableRow key={b.id} className="border-white/5">
   323→   323→                      <TableCell className="max-w-[180px] truncate font-medium text-zinc-200" title={b.name}>
   324→   324→                        {b.name}
   325→   325→                      </TableCell>
   326→   326→                      <TableCell className="num whitespace-nowrap text-zinc-300">
   327→   327→                        {formatBytes(b.sizeBytes)}
   328→   328→                      </TableCell>
   329→   329→                      <TableCell className="num whitespace-nowrap text-zinc-400">
   330→   330→                        {formatTime(b.createdAt)}
   331→   331→                      </TableCell>
   332→   332→                      <TableCell>
   333→   333→                        {b.uploaded ? (
   334→   334→                          <Badge className="rounded-full border-emerald-500/40 bg-emerald-500/10 text-emerald-400">
   335→   335→                            Đã lên
   336→   336→                          </Badge>
   337→   337→                        ) : (
   338→   338→                          <Badge variant="outline" className="rounded-full border-white/10 text-zinc-500">
   339→   339→                            Chưa
   340→   340→                          </Badge>
   341→   341→                        )}
   342→   342→                      </TableCell>
   343→   343→                      <TableCell className="text-right">
   344→   344→                        <div className="flex justify-end gap-1">
   345→   345→                          <Button
   346→   346→                            size="sm"
   347→   347→                            variant="ghost"
   348→   348→                            title={b.uploaded ? 'Đã tải lên bucket' : 'Tải lên bucket'}
   349→   349→                            disabled={b.uploaded || busyLocalId === b.id}
   350→   350→                            className="h-8 rounded-lg px-2 text-zinc-300 hover:text-white disabled:opacity-40"
   351→   351→                            onClick={() => uploadOne(b)}
   352→   352→                          >
   353→   353→                            {busyLocalId === b.id ? (
   354→   354→                              <Loader2 className="h-3.5 w-3.5 animate-spin" />
   355→   355→                            ) : (
   356→   356→                              <UploadCloud className="h-3.5 w-3.5" />
   357→   357→                            )}
   358→   358→                          </Button>
   359→   359→                          <Button
   360→   360→                            size="sm"
   361→   361→                            variant="ghost"
   362→   362→                            title="Tải về (link pre-signed)"
   363→   363→                            disabled={!b.uploaded || busyLocalId === b.id}
   364→   364→                            className="h-8 rounded-lg px-2 text-zinc-300 hover:text-white disabled:opacity-40"
   365→   365→                            onClick={() => downloadLocal(b)}
   366→   366→                          >
   367→   367→                            <Download className="h-3.5 w-3.5" />
   368→   368→                          </Button>
   369→   369→                          <AlertDialog>
   370→   370→                            <AlertDialogTrigger asChild>
   371→   371→                              <Button
   372→   372→                                size="sm"
   373→   373→                                variant="ghost"
   374→   374→                                title="Xóa bản backup này"
   375→   375→                                disabled={busyLocalId === b.id}
   376→   376→                                className="h-8 rounded-lg px-2 text-rose-400 hover:text-rose-300"
   377→   377→                              >
   378→   378→                                <Trash2 className="h-3.5 w-3.5" />
   379→   379→                              </Button>
   380→   380→                            </AlertDialogTrigger>
   381→   381→                            <AlertDialogContent className="glass rounded-2xl border-white/10">
   382→   382→                              <AlertDialogHeader>
   383→   383→                                <AlertDialogTitle className="text-zinc-100">
   384→   384→                                  Xóa bản backup trên máy chủ?
   385→   385→                                </AlertDialogTitle>
   386→   386→                                <AlertDialogDescription className="text-zinc-400">
   387→   387→                                  File <span className="num">{b.name}</span> sẽ bị xóa vĩnh viễn khỏi

   389→   389→                                </AlertDialogDescription>
   390→   390→                              </AlertDialogHeader>
   391→   391→                              <AlertDialogFooter>
   392→   392→                                <AlertDialogCancel className="rounded-xl border-white/10 bg-transparent text-zinc-200 hover:bg-white/10">
   393→   393→                                  Hủy
   394→   394→                                </AlertDialogCancel>
   395→   395→                                <AlertDialogAction
   396→   396→                                  className="rounded-xl bg-rose-600 text-white hover:bg-rose-500"
   397→   397→                                  onClick={() => deleteLocal(b.id)}
   398→   398→                                >
   399→   399→                                  Xóa
   400→   400→                                </AlertDialogAction>
   401→   401→                              </AlertDialogFooter>
   402→   402→                            </AlertDialogContent>
   403→   403→                          </AlertDialog>
   404→   404→                        </div>
   405→   405→                      </TableCell>
   406→   406→                    </TableRow>
   407→   407→                  ))}
   408→   408→                </TableBody>
   409→   409→              </Table>
   410→   410→            </div>
   411→   411→          )}
   412→   412→        </Card>
   413→   413→
   414→   414→        {/* Bucket */}
   415→   415→        <Card className="glass rounded-2xl p-4">
