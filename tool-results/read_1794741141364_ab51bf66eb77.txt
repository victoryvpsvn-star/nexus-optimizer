    46→    46→type ListData = BackupListResponse & { totalLocalBytes?: number }
    47→    47→
    48→    48→export function BackupPanel() {
    49→    49→  const [data, setData] = useState<ListData | null>(null)
    50→    50→  const [loading, setLoading] = useState(true)
    51→    51→  const [name, setName] = useState('')
    52→    52→  const [creating, setCreating] = useState(false)
    53→    53→  const [busyLocalId, setBusyLocalId] = useState<number | null>(null)
    54→    54→  const [busyBucketKey, setBusyBucketKey] = useState<string | null>(null)
    55→    55→
    56→    56→  const local = data?.local ?? []
    57→    57→  const bucket = data?.bucket ?? []
    58→    58→  const totalLocalBytes = data?.totalLocalBytes ?? local.reduce((s, b) => s + b.sizeBytes, 0)
    59→    59→  const totalBucketBytes = bucket.reduce((s, b) => s + b.sizeBytes, 0)
    60→    60→
    61→    61→  const load = useCallback(async () => {
    62→    62→    try {
    63→    63→      const r = await fetch('/api/backups', { cache: 'no-store' })
    64→    64→      const json = (await r.json().catch(() => null)) as (ListData & { error?: string }) | null
    65→    65→      if (!r.ok || !json || json.error) throw new Error(json?.error ?? 'Không tải được danh sách backup')
    66→    66→      setData(json)
    67→    67→    } catch (e) {
    68→    68→      /* giữ danh sách cũ khi lỗi — không làm trống bảng */
    69→    69→      toast.error((e as Error).message || 'Không tải được danh sách backup')
    70→    70→    } finally {
    71→    71→      setLoading(false)
    72→    72→    }
    73→    73→  }, [])
    74→    74→
    75→    75→  useEffect(() => {
    76→    76→    load()
    77→    77→  }, [load])
    78→    78→
    79→    79→  const create = async () => {
    80→    80→    setCreating(true)
    81→    81→    try {
    82→    82→      const r = await fetch('/api/backups', {
    83→    83→        method: 'POST',
    84→    84→        headers: { 'Content-Type': 'application/json' },
    85→    85→        body: JSON.stringify({ name: name.trim() || 'backup' }),
    86→    86→      })
    87→    87→      const json = (await r.json()) as { backup?: BackupLocal; error?: string }
    88→    88→      if (!r.ok || json.error) throw new Error(json.error || 'Tạo backup thất bại')
    89→    89→      toast.success(
    90→    90→        `Đã tạo backup ${json.backup?.name ?? ''} (${formatBytes(json.backup?.sizeBytes ?? 0)})`,
    91→    91→      )
    92→    92→      setName('')
    93→    93→      await load()
    94→    94→    } catch (e) {
    95→    95→      toast.error((e as Error).message || 'Tạo backup thất bại')
    96→    96→    } finally {
    97→    97→      setCreating(false)
    98→    98→    }
    99→    99→  }
   100→   100→
   101→   101→  const uploadOne = async (b: BackupLocal) => {
   102→   102→    setBusyLocalId(b.id)
   103→   103→    toast.info(`Đang tải ${b.name} lên bucket...`)
   104→   104→    try {
   105→   105→      const r = await fetch('/api/backups', {
   106→   106→        method: 'PUT',
   107→   107→        headers: { 'Content-Type': 'application/json' },
   108→   108→        body: JSON.stringify({ action: 'upload', id: b.id }),
   109→   109→      })
   110→   110→      const json = (await r.json()) as { ok?: boolean; error?: string; key?: string }
   111→   111→      if (!r.ok || !json.ok) throw new Error(json.error || 'Tải lên bucket thất bại')
   112→   112→      toast.success(`Đã tải lên bucket: ${json.key ?? b.name}`)
   113→   113→      await load()
   114→   114→    } catch (e) {
   115→   115→      toast.error((e as Error).message || 'Tải lên bucket thất bại')
   116→   116→    } finally {
   117→   117→      setBusyLocalId(null)
   118→   118→    }
   119→   119→  }
   120→   120→
   121→   121→  const restoreLocal = async (b: BackupLocal) => {
   122→   122→    setBusyLocalId(b.id)
   123→   123→    toast.info('Đang dùng lại backup từ file local...')
   124→   124→    try {
   125→   125→      const r = await fetch('/api/backups', {
   126→   126→        method: 'PUT',
   127→   127→        headers: { 'Content-Type': 'application/json' },
   128→   128→        body: JSON.stringify({ action: 'restore-local', id: b.id }),
   129→   129→      })
   130→   130→      const json = (await r.json()) as { ok?: boolean; message?: string; error?: string }
   131→   131→      if (!r.ok || !json.ok) throw new Error(json.error || 'Khôi phục thất bại')
   132→   132→      toast.success(json.message || 'Đã khôi phục backup từ máy chủ')
   133→   133→    } catch (e) {
   134→   134→      toast.error((e as Error).message || 'Khôi phục thất bại')
   135→   135→    } finally {
   136→   136→      setBusyLocalId(null)
   137→   137→    }
   138→   138→  }
   139→   139→
   140→   140→  const downloadLocal = async (b: BackupLocal) => {
   141→   141→    setBusyLocalId(b.id)
   142→   142→    try {
   143→   143→      const r = await fetch('/api/backups', {
   144→   144→        method: 'PUT',
   145→   145→        headers: { 'Content-Type': 'application/json' },
   146→   146→        body: JSON.stringify({ action: 'presign', id: b.id }),
   147→   147→      })
   148→   148→      const json = (await r.json()) as { url?: string; error?: string }
   149→   149→      if (!r.ok || !json.url) throw new Error(json.error || 'Không tạo được link tải về')
   150→   150→      window.open(json.url, '_blank', 'noopener')
   151→   151→      toast.success('Đã mở link tải về (hết hạn sau 1 giờ)')
   152→   152→    } catch (e) {
   153→   153→      toast.error((e as Error).message || 'Không tạo được link tải về')
   154→   154→    } finally {
   155→   155→      setBusyLocalId(null)
   156→   156→    }
   157→   157→  }
   158→   158→
   159→   159→  const downloadBucket = async (key: string) => {
   160→   160→    setBusyBucketKey(key)
   161→   161→    try {
   162→   162→      const r = await fetch('/api/backups', {
   163→   163→        method: 'PUT',
   164→   164→        headers: { 'Content-Type': 'application/json' },
   165→   165→        body: JSON.stringify({ action: 'presign', key }),
   166→   166→      })
   167→   167→      const json = (await r.json()) as { url?: string; error?: string }
   168→   168→      if (!r.ok || !json.url) throw new Error(json.error || 'Không tạo được link tải về')
   169→   169→      window.open(json.url, '_blank', 'noopener')
   170→   170→      toast.success('Đã mở link tải về (hết hạn sau 1 giờ)')
   171→   171→    } catch (e) {
   172→   172→      toast.error((e as Error).message || 'Không tạo được link tải về')
   173→   173→    } finally {
   174→   174→      setBusyBucketKey(null)
   175→   175→    }
   176→   176→  }
   177→   177→
   178→   178→  const restoreBucket = async (key: string) => {
   179→   179→    setBusyBucketKey(key)
   180→   180→    toast.info('Đang khôi phục backup về máy chủ...')
   181→   181→    try {
   182→   182→      const r = await fetch('/api/backups', {
   183→   183→        method: 'PUT',
   184→   184→        headers: { 'Content-Type': 'application/json' },
   185→   185→        body: JSON.stringify({ action: 'restore', key }),
   186→   186→      })
   187→   187→      const json = (await r.json()) as { ok?: boolean; message?: string; error?: string }
   188→   188→      if (!r.ok || !json.ok) throw new Error(json.error || 'Khôi phục thất bại')
   189→   189→      toast.success(json.message || 'Đã khôi phục backup về máy chủ')
   190→   190→    } catch (e) {
   191→   191→      toast.error((e as Error).message || 'Khôi phục thất bại')
   192→   192→    } finally {
   193→   193→      setBusyBucketKey(null)
   194→   194→    }
   195→   195→  }
   196→   196→
   197→   197→  const deleteLocal = async (id: number) => {
   198→   198→    setBusyLocalId(id)
   199→   199→    try {
   200→   200→      const r = await fetch(`/api/backups?type=local&id=${id}`, { method: 'DELETE' })
   201→   201→      const json = (await r.json()) as { success?: boolean; error?: string }
   202→   202→      if (!r.ok || !json.success) throw new Error(json.error || 'Không xóa được bản backup')
   203→   203→      toast.success('Đã xóa bản backup trên máy chủ')
   204→   204→      await load()
   205→   205→    } catch (e) {
   206→   206→      toast.error((e as Error).message || 'Không xóa được bản backup')
   207→   207→    } finally {
   208→   208→      setBusyLocalId(null)
   209→   209→    }
   210→   210→  }
   211→   211→
   212→   212→  const deleteBucket = async (key: string) => {
   213→   213→    setBusyBucketKey(key)
   214→   214→    try {
   215→   215→      const r = await fetch(`/api/backups?type=bucket&key=${encodeURIComponent(key)}`, {
   216→   216→        method: 'DELETE',
   217→   217→      })
   218→   218→      const json = (await r.json()) as { success?: boolean; error?: string }
   219→   219→      if (!r.ok || !json.success) throw new Error(json.error || 'Không xóa được object trên bucket')
   220→   220→      toast.success('Đã xóa object trên bucket')
   221→   221→      await load()
   222→   222→    } catch (e) {
   223→   223→      toast.error((e as Error).message || 'Không xóa được object trên bucket')
   224→   224→    } finally {
   225→   225→      setBusyBucketKey(null)
   226→   226→    }
   227→   227→  }
   228→   228→
   229→   229→  return (
   230→   230→    <div className="space-y-4">
   231→   231→      {/* Hero: tạo backup + trạng thái */}
   232→   232→      <Card className="glass rounded-2xl p-6">
   233→   233→        <div className="flex flex-col gap-5">
   234→   234→          <div className="flex flex-wrap items-center justify-between gap-4">
   235→   235→            <div className="flex items-center gap-3">
   236→   236→              <div
   237→   237→                className="flex h-11 w-11 items-center justify-center rounded-xl border border-white/10"
   238→   238→                style={{ background: 'color-mix(in srgb, var(--accent-1) 14%, transparent)' }}
   239→   239→              >
   240→   240→                <DatabaseBackup className="h-5 w-5" style={{ color: 'var(--accent-1)' }} />
   241→   241→              </div>
   242→   242→              <div>
   243→   243→                <h3 className="text-base font-semibold text-zinc-100">Tạo bản sao lưu</h3>
   244→   244→                <p className="text-xs text-zinc-500">
   245→   245→                  Nén toàn bộ mã nguồn + cấu hình thành file .tar.gz (loại trừ node_modules,
   246→   246→                  .next, upload, .env)
   247→   247→                </p>
   248→   248→              </div>
   249→   249→            </div>
   250→   250→            <div className="flex flex-wrap items-center gap-2">
   251→   251→              {data ? (
   252→   252→                data.bucketOk ? (
   253→   253→                  <Badge className="rounded-full border-emerald-500/40 bg-emerald-500/10 text-emerald-400">
   254→   254→                    <ShieldCheck className="mr-1 h-3.5 w-3.5" />
   255→   255→                    Bucket <span className="num mx-1">{data.bucketName ?? 'vic'}</span> sẵn sàng
   256→   256→                  </Badge>
   257→   257→                ) : (
   258→   258→                  <Badge className="rounded-full border-rose-500/40 bg-rose-500/10 text-rose-400">
   259→   259→                    <TriangleAlert className="mr-1 h-3.5 w-3.5" />
   260→   260→                    Bucket lỗi: {data.bucketError ?? 'không kết nối được'}
   261→   261→                  </Badge>
   262→   262→                )
   263→   263→              ) : (
   264→   264→                <Skeleton className="h-6 w-44 rounded-full" />
   265→   265→              )}
   266→   266→              <Badge variant="outline" className="rounded-full border-white/10 text-zinc-300">
   267→   267→                <HardDrive className="mr-1 h-3.5 w-3.5" style={{ color: 'var(--accent-2)' }} />
   268→   268→                Dung lượng local: <span className="num ml-1">{formatBytes(totalLocalBytes)}</span>
   269→   269→              </Badge>
   270→   270→              <Badge variant="outline" className="rounded-full border-white/10 text-zinc-300">
   271→   271→                <Cloud className="mr-1 h-3.5 w-3.5" style={{ color: 'var(--accent-1)' }} />
   272→   272→                Dung lượng bucket: <span className="num ml-1">{formatBytes(totalBucketBytes)}</span>
   273→   273→                <span className="num ml-1 text-zinc-500">/ 30GB</span>
   274→   274→              </Badge>
   275→   275→            </div>
   276→   276→          </div>
   277→   277→
   278→   278→          <div className="flex flex-col gap-3 sm:flex-row">
   279→   279→            <Input
   280→   280→              value={name}
   281→   281→              onChange={(e) => setName(e.target.value)}
   282→   282→              onKeyDown={(e) => {
   283→   283→                if (e.key === 'Enter' && !creating) create()
   284→   284→              }}
   285→   285→              placeholder="vd: nexus-full-2025"
   286→   286→              className="h-11 flex-1 rounded-xl border-white/10 bg-white/5 text-zinc-100 placeholder:text-zinc-400"
   287→   287→              aria-label="Tên bản sao lưu"
   288→   288→            />
   289→   289→            <Button
   290→   290→              onClick={create}
   291→   291→              disabled={creating}
   292→   292→              className="h-11 rounded-xl px-6 font-semibold text-black"
   293→   293→              style={{ background: 'var(--accent-1)' }}
   294→   294→            >
   295→   295→              {creating ? (
   296→   296→                <Loader2 className="mr-1.5 h-4 w-4 animate-spin" />
   297→   297→              ) : (
   298→   298→                <DatabaseBackup className="mr-1.5 h-4 w-4" />
   299→   299→              )}
   300→   300→              {creating ? 'Đang tạo backup...' : 'Tạo backup ngay'}
   301→   301→            </Button>
   302→   302→          </div>
   303→   303→        </div>
   304→   304→      </Card>
   305→   305→
   306→   306→      {/* Hai danh sách */}
   307→   307→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-2">
   308→   308→        {/* Local */}
   309→   309→        <Card className="glass rounded-2xl p-4">
   310→   310→          <h3 className="mb-3 flex items-center gap-2 text-sm font-semibold text-zinc-200">
   311→   311→            <HardDrive className="h-4 w-4" style={{ color: 'var(--accent-2)' }} />
   312→   312→            Bản sao lưu trên máy chủ
   313→   313→            <Badge variant="outline" className="num ml-auto rounded-full border-white/10 text-zinc-400">
   314→   314→              {loading ? '...' : local.length}
   315→   315→            </Badge>
   316→   316→          </h3>
   317→   317→          {loading ? (
   318→   318→            <div className="space-y-2">
   319→   319→              <Skeleton className="h-10 w-full rounded-xl" />
   320→   320→              <Skeleton className="h-10 w-full rounded-xl" />
   321→   321→            </div>
   322→   322→          ) : local.length === 0 ? (
   323→   323→            <div className="flex flex-col items-center gap-2 py-10 text-center">
   324→   324→              <PackageOpen className="h-8 w-8 text-zinc-400" />
   325→   325→              <p className="text-sm text-zinc-500">
   326→   326→                Chưa có bản backup nào trên máy chủ — tạo bản đầu tiên ở trên.
   327→   327→              </p>
   328→   328→            </div>
   329→   329→          ) : (
   330→   330→            <div className="thin-scroll max-h-96 overflow-y-auto">
   331→   331→              <Table>
   332→   332→                <TableHeader>
   333→   333→                  <TableRow className="border-white/10">
   334→   334→                    <TableHead className="text-zinc-400">Tên</TableHead>
   335→   335→                    <TableHead className="text-zinc-400">Dung lượng</TableHead>
   336→   336→                    <TableHead className="text-zinc-400">Thời gian</TableHead>
   337→   337→                    <TableHead className="text-zinc-400">Bucket</TableHead>
   338→   338→                    <TableHead className="text-right text-zinc-400">Hành động</TableHead>
   339→   339→                  </TableRow>
   340→   340→                </TableHeader>
   341→   341→                <TableBody>
   342→   342→                  {local.map((b) => (
   343→   343→                    <TableRow key={b.id} className="border-white/5">
   344→   344→                      <TableCell className="max-w-[180px] truncate font-medium text-zinc-200" title={b.name}>
   345→   345→                        {b.name}
   346→   346→                      </TableCell>
   347→   347→                      <TableCell className="num whitespace-nowrap text-zinc-300">
   348→   348→                        {formatBytes(b.sizeBytes)}
   349→   349→                      </TableCell>
   350→   350→                      <TableCell className="num whitespace-nowrap text-zinc-400">
   351→   351→                        {formatTime(b.createdAt)}
   352→   352→                      </TableCell>
   353→   353→                      <TableCell>

   355→   355→                          <Badge className="rounded-full border-emerald-500/40 bg-emerald-500/10 text-emerald-400">
   356→   356→                            Đã tải lên
   357→   357→                          </Badge>
   358→   358→                        ) : (
   359→   359→                          <Badge variant="outline" className="rounded-full border-white/10 text-zinc-500">
   360→   360→                            Chưa tải lên
   361→   361→                          </Badge>
   362→   362→                        )}
   363→   363→                      </TableCell>
   364→   364→                      <TableCell className="text-right">
   365→   365→                        <div className="flex justify-end gap-1">
   366→   366→                          <Button

   372→   372→                            className="h-9 rounded-lg px-2.5 text-zinc-300 hover:text-white disabled:opacity-40"
   373→   373→                            onClick={() => uploadOne(b)}
   374→   374→                          >
   375→   375→                            {busyLocalId === b.id ? (
   376→   376→                              <Loader2 className="h-3.5 w-3.5 animate-spin" />
   377→   377→                            ) : (
   378→   378→                              <UploadCloud className="h-3.5 w-3.5" />
   379→   379→                            )}
   380→   380→                          </Button>
   381→   381→                          <Button
   382→   382→                            size="sm"
   383→   383→                            variant="ghost"
   384→   384→                            title="Tải về (link pre-signed)"
   385→   385→                            aria-label={`Tải về bản backup ${b.name}`}
   386→   386→                            disabled={!b.uploaded || busyLocalId === b.id}
   387→   387→                            className="h-9 rounded-lg px-2.5 text-zinc-300 hover:text-white disabled:opacity-40"
   388→   388→                            onClick={() => downloadLocal(b)}
   389→   389→                          >
   390→   390→                            <Download className="h-3.5 w-3.5" />
   391→   391→                          </Button>
   392→   392→                          <Button
   393→   393→                            size="sm"
   394→   394→                            variant="ghost"
   395→   395→                            title="Dùng lại backup này (khôi phục về máy chủ)"
   396→   396→                            aria-label={`Dùng lại backup ${b.name}`}
   397→   397→                            disabled={busyLocalId === b.id}
   398→   398→                            className="h-9 rounded-lg px-2.5 text-zinc-300 hover:text-white disabled:opacity-40"
   399→   399→                            onClick={() => restoreLocal(b)}
   400→   400→                          >
   401→   401→                            {busyLocalId === b.id ? (
   402→   402→                              <Loader2 className="h-3.5 w-3.5 animate-spin" />
   403→   403→                            ) : (
   404→   404→                              <RotateCcw className="h-3.5 w-3.5" />
